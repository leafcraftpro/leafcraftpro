# ============================================================
#  LeafCraftPRO — Apache configuration for the NODE.JS deployment.
#
#  This is only needed if you choose "Setup Node.js App" in hPanel
#  instead of the static upload. Hostinger normally writes most of
#  this for you; the file is here so you can recover or adjust it.
#
#  Replace APP_PATH below with the absolute path to this folder,
#  for example: /home/u123456789/domains/leafcraftpro.site/app
# ============================================================

PassengerAppRoot "/home/USERNAME/domains/leafcraftpro.site/app"
PassengerBaseURI "/"
PassengerNodejs "/home/USERNAME/nodevenv/domains/leafcraftpro.site/app/22/bin/node"
PassengerAppType node
PassengerStartupFile server.js

# Long-running requests (image-heavy first paint, slow clients).
PassengerFriendlyErrorPages off

# ---------- Compression ----------
<IfModule mod_deflate.c>
    <IfModule mod_filter.c>
        AddOutputFilterByType DEFLATE text/plain text/html text/xml text/css
        AddOutputFilterByType DEFLATE application/javascript application/json
        AddOutputFilterByType DEFLATE application/ld+json image/svg+xml
    </IfModule>
</IfModule>

# ---------- Caching ----------
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType text/css                "access plus 1 year"
    ExpiresByType application/javascript  "access plus 1 year"
    ExpiresByType image/webp              "access plus 1 year"
    ExpiresByType image/svg+xml           "access plus 1 year"
    ExpiresByType font/woff2              "access plus 1 year"
</IfModule>

# ---------- Security ----------
<IfModule mod_headers.c>
    Header always set X-Content-Type-Options "nosniff"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
    Header unset X-Powered-By
</IfModule>
